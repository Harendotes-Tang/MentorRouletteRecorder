# mr_desktop_suite_source(<Suite> <var>): <Suite>Tests.cpp in whichever area folder holds it.
function(mr_desktop_suite_source suite out_var)
    foreach(area IN LISTS MR_DESKTOP_TEST_AREAS)
        if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${area}/${suite}Tests.cpp")
            set(${out_var} "${area}/${suite}Tests.cpp" PARENT_SCOPE)
            return()
        endif()
    endforeach()
    message(FATAL_ERROR "${suite}Tests.cpp is in none of the test area folders: ${MR_DESKTOP_TEST_AREAS}")
endfunction()

# One test executable per Desktop area. Each suite's former main() is an entry function
# (support/DesktopTestRunner.h); the area executable links the suites of its area and runs the
# one its first argument names, one suite per process, so every suite keeps its own application
# object, environment and exit codes. Linking a dozen executables instead of one per suite is
# most of what a Debug build of the tests costs. CTest still registers, names and logs every
# suite on its own (mr_desktop_suite_test).

# mr_desktop_test_executable(<target> SUITES <Suite>... [LIBRARIES <lib>...] [DEFINITIONS <def>...])
function(mr_desktop_test_executable target)
    cmake_parse_arguments(PARSE_ARGV 1 ARG "" "" "SUITES;LIBRARIES;DEFINITIONS")
    set(sources support/DesktopTestMain.cpp support/DesktopTestRunner.h)
    set(declarations "")
    set(entries "")
    foreach(suite IN LISTS ARG_SUITES)
        mr_desktop_suite_source(${suite} source)
        list(APPEND sources "${source}")
        string(APPEND declarations "int mrDesktopSuite_${suite}(int argc, char **argv);\n")
        string(APPEND entries "    {\"${suite}\", &mrDesktopSuite_${suite}},\n")
    endforeach()
    # The suite table is written from the same list, so a listed suite without an entry fails to link.
    set(generated "${CMAKE_CURRENT_BINARY_DIR}/${target}-suites")
    file(CONFIGURE OUTPUT "${generated}/DesktopTestSuites.g.h" CONTENT
"// Generated by tests/Desktop.Tests/CMakeLists.txt: the suites of this test executable.
#pragma once

@declarations@
namespace mrtest {

inline constexpr SuiteEntry Suites[] = {
@entries@};

} // namespace mrtest
" @ONLY)
    add_executable(${target} ${sources})
    target_include_directories(${target} PRIVATE "${generated}")
    target_link_libraries(${target} PRIVATE MentorRecorderDesktopLib Qt6::Gui Qt6::Test ${ARG_LIBRARIES})
    target_compile_definitions(${target} PRIVATE
        MR_SOURCE_DIR="${CMAKE_SOURCE_DIR}"
        MR_DESKTOP_QML_DIR="${PROJECT_SOURCE_DIR}/src/Desktop/qml"
        ${ARG_DEFINITIONS})
    set_target_properties(${target} PROPERTIES AUTOMOC TRUE)
    mr_use_ascii_autogen_dir(${target})
endfunction()

# mr_desktop_suite_test(<target> <Suite> [NAME <test>] [LOG <file> | NO_LOG] [ARGS <arg>...])
# One CTest entry: runs <Suite> of <target>. The name defaults to MentorRecorder<Suite> and the
# QtTest log to <name>.qtlog.txt beside the executables, as when every suite had its own binary.
function(mr_desktop_suite_test target suite)
    cmake_parse_arguments(PARSE_ARGV 2 ARG "NO_LOG" "NAME;LOG" "ARGS")
    if(NOT ARG_NAME)
        set(ARG_NAME "MentorRecorder${suite}")
    endif()
    if(NOT ARG_LOG)
        set(ARG_LOG "${CMAKE_CURRENT_BINARY_DIR}/${ARG_NAME}.qtlog.txt")
    endif()
    set(command ${target} ${suite} ${ARG_ARGS})
    if(NOT ARG_NO_LOG)
        list(APPEND command -o "${ARG_LOG},txt")
    endif()
    add_test(NAME ${ARG_NAME} COMMAND ${command})
endfunction()

# ---------------------------------------------------------------------------
# Keep process-supervision tests away from the user's Collector and database.
#
# AppController has no process authority unless a supervisor is explicitly
# injected, and tests inject one only with a stub they name themselves. MR_DATA_DIR
# keeps lease/takeover inspection away from the user's live serve.pid.
# MR_COLLECTOR_PATH matters only to a build compiled with
# MR_DEV_COLLECTOR_DISCOVERY, where it is authoritative and exclusive; a release
# build never reads it. Supervision fixtures override these values only with
# their own temporary stubs and directories.
#
# MentorRecorderIpcIntegration is included too: it launches its fixture
# Collector with a UUID pipe, temporary database/log paths and its own
# MR_DATA_DIR in the child environment, so these defaults protect the test
# process without changing the fixture child's setup.
set_property(TEST ${MR_ALL_DESKTOP_TESTS} APPEND PROPERTY ENVIRONMENT
    "MR_COLLECTOR_PATH=${CMAKE_CURRENT_BINARY_DIR}/no-such-collector/MentorRecorder.Collector.exe"
    "MR_DATA_DIR=${CMAKE_CURRENT_BINARY_DIR}/no-collector-data")

# A QML or JavaScript runtime diagnostic names the QML file it comes from
# (qrc:/qt/qml/MentorRecorder/... or file:///...). The screenshot harness prints
# it to stderr ("[warning] <url>:<line>: TypeError: ...", main.cpp's message
# handler) and carries on to the frame, so a binding that throws, a failed
# property assignment or an unresolved name would otherwise still pass (review
# OJ-7). The harness counts them itself and exits 10 (QmlWarningCounter, review
# DT4-X5); this pattern stays as a second net, and it also sees a warning logged
# while the window is torn down. The harness's two environment notices (no font
# directory, no Direct3D 12) name no QML file and are not matched. No "\\[" in front: CTest's regex
# engine then fails to match the group that follows it.
set(MR_QML_WARNING_REGEX "(warning|critical|fatal)\\] (qrc|file):")
set_tests_properties(${MR_QML_TESTS} ${MR_VALIDATION_QML_TESTS} ${MR_RECORDING_QML_TESTS}
    PROPERTIES FAIL_REGULAR_EXPRESSION "${MR_QML_WARNING_REGEX}")
